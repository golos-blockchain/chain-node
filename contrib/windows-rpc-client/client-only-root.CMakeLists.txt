cmake_minimum_required(VERSION 3.10)
project(GolosWindowsClient C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wa,-mbig-obj")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static")
set(CPP_STANDARD -std=c++14)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/thirdparty/fc/CMakeModules" "${CMAKE_SOURCE_DIR}/thirdparty/fc/GitVersionGen")
add_subdirectory(thirdparty/fc)
add_subdirectory(libraries/utilities)
add_subdirectory(libraries/protocol)
add_subdirectory(libraries/network)
add_subdirectory(programs/rpc_directory_client)
